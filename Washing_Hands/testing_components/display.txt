void show_page() {
    // 1. Pinta o ecrã de Preto para limpar tudo
    tft.fillScreen(ST77XX_BLACK);
    
    // 2. Desenha a Imagem
    // Como o teu ecrã tem 160 de largura e a imagem tem 64, o meio exato é X = 48
    // Como o ecrã tem 128 de altura e a imagem tem 64, o meio é Y = 32
    // tft.drawBitmap(X, Y, nome_da_imagem, Largura, Altura, Cor_do_Desenho);
    tft.drawBitmap(48, 10, smile_bitmap_imagem, 64, 64, ST77XX_WHITE);
    
    // 3. Escreve o teu texto por baixo da imagem!
    tft.setTextColor(ST77XX_WHITE);
    tft.setTextSize(1);
    tft.setFont(); // Usar a fonte básica e pequenina
    tft.setCursor(50, 85); // Posição X e Y do texto
    tft.print("O Abacate");
    tft.setCursor(65, 100);
    tft.print("PBL1");
}

void setup(void) {
    Serial.begin(115200);
    tft.initR(INITR_BLACKTAB); 
    
    // Rodar para horizontal
    tft.setRotation(1); 
    
    show_page();
}
